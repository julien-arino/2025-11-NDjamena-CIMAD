\begin{frame}
Voir en particulier les travaux de \href{https://scholar.google.ca/citations?user=o7R6ZHMAAAAJ}{Horst Thieme}
\vfill
Si l'on considère le temps de séjour dans les compartiments d'une perspective plus détaillée, on obtient des modèles intégro-différentiels
\vfill
Nous utilisons ici des variables aléatoires continues. Voir les chapitres 12 et 13 du \href{https://press.princeton.edu/books/paperback/9780691092911/mathematics-in-population-biology}{livre de Thieme} pour des distributions arbitraires
\end{frame}


\begin{frame}\frametitle{Temps jusqu'aux événements}
Nous supposons qu'un système peut être dans deux états, $A$ et $B$
\begin{itemize}
\item Au temps $t=0$, le système est dans l'état $A$
\item Un événement se produit à un certain temps $t=\tau$, qui déclenche le passage de
l'état $A$ à l'état $B$
\end{itemize}
\vfill
Appelons $T$ la variable aléatoire 
\begin{quote}
``temps passé dans l'état $A$ avant de passer à l'état $B$''
\end{quote}
\end{frame}

\begin{frame}
Les états peuvent être n'importe quoi :
\begin{itemize}
\item $A$ : fonctionnel, $B$ : en panne
\item $A$ : infecté, $B$ : rétabli
\item $A$ : vivant, $B$ : mort
\item $\ldots$
\end{itemize}
\vfill
Nous prenons une collection d'objets ou d'individus qui sont dans l'état $A$ et voulons
une loi pour la \defword{distribution} des temps passés dans $A$, c'est-à-dire une loi pour $T$
\vfill
Par exemple, nous fabriquons des ampoules et aimerions dire à nos clients qu'en moyenne, nos ampoules durent 200 ans...
\vfill
Nous effectuons un nombre \defword{infini} d'expériences, et observons le temps qu'il faut, dans chaque expérience, pour passer de $A$ à $B$
\end{frame}

\begin{frame}\frametitle{Une distribution de probabilité est un modèle}
À partir de la séquence d'expériences, nous déduisons un modèle, qui dans ce contexte est appelé une \defword{distribution de probabilité}
\vfill
Nous supposons que $T$ est une variable aléatoire \defword{continue}
\end{frame}



\begin{frame}\frametitle{Fonction de densité de probabilité}
Puisque $T$ est continue, elle a une \defword{fonction de densité de probabilité} continue $f$
\begin{minipage}{0.4\textwidth}
\begin{itemize}
\item $f\geq 0$
\item $\int_{-\infty}^{+\infty}f(s)ds=1$
\item $\IP(a\leq T\leq b)=\int_a^bf(t)dt$
\end{itemize}
\end{minipage}
\begin{minipage}{0.59\textwidth}
\includegraphics[width=\textwidth]{FIGS/course-03-distrib_a_b-1.pdf}
\end{minipage}
\end{frame}



\begin{frame}\frametitle{Fonction de distribution cumulative (f.d.c.)}
\begin{minipage}{0.4\textwidth}
La \defword{fonction de distribution cumulative} est une fonction $F(t)$ qui caractérise la distribution de $T$, et définie par
\[
F(s)=\IP(T\leq s)=\int_{-\infty}^sf(x)dx
\]
\end{minipage}
\begin{minipage}{0.59\textwidth}
\includegraphics[width=\textwidth]{FIGS/course-03-distrib_minf_b-1.pdf}
\end{minipage}
\end{frame}

\begin{frame}\frametitle{Fonction de survie}
Une autre caractérisation de la distribution de la variable aléatoire $T$ se fait par la fonction de \defword{survie} (ou de \defword{séjour})
\vfill
La fonction de survie de l'état $A$ est donnée par 
\begin{equation}
  \S(t)=1-F(t)=\IP(T>t)
  \label{eq:survival}
\end{equation}
Ceci donne une description du \defword{temps de séjour} d'un
système dans un état particulier (le temps passé dans l'état)
\vfill
$\S$ est une fonction non croissante (puisque $\S=1-F$
avec $F$ une f.d.c.), et
$\S(0)=1$ (puisque $T$ est une variable aléatoire non négative)
\end{frame}



\maxFrameImage{FIGS/course-03-pdf-cdf-surv-normal-1.pdf}

\begin{frame}
Le \defword{temps de séjour moyen} $\tau$ dans l'état $A$ est donné par
\[
\tau=E(T)=\int_0^\infty tf(t)dt
\]
Puisque $\lim_{t\to\infty}t\S(t)=0$, il s'ensuit que 
\[
\tau=\int_0^\infty \S(t)dt
\]
\vfill
\defword{Durée de vie future espérée} :
\[
\frac{1}{\S(t_0)} \int_0^{\infty} t\,f(t+t_0)\,dt 
\]
\vfill
\begin{eqnarray*}
\S(t)-\S(a)&=&\IP\left\{\textrm{survivre pendant }
 (a,t)\textrm{ ayant survécu jusqu'à }a\right\} \\
&=& \exp\left(-\int_a^t h(u)du\right)
\end{eqnarray*}
\end{frame}

\begin{frame}\frametitle{Taux de risque}
Le \defword{taux de risque} (ou \defword{taux de défaillance}) est
\begin{align*}
h(t) &= \lim_{\Delta t\to 0}\frac{\S(t)-\S(t+\Delta t)}{\Delta t} \\
& = \lim_{\Delta t\to 0} \frac{\IP(T<t+\Delta t | T\geq
t)}{\Delta t} \\
&= \frac{f(t)}{\S(t)}
\end{align*}
Il donne la probabilité de défaillance entre $t$ et $\Delta t$, étant donné la survie jusqu'à $t$.
\vfill
Nous avons
\[
h(t)=-\frac{d}{dt}\ln\S(t)
\]
\end{frame}

\begin{frame}{Risques concurrents}
Supposons maintenant que le système commence dans l'état $A$ au temps $t=0$ et que selon lequel des deux événements $\mathcal{E}_1$ ou $\mathcal{E}_2$ a lieu en premier, il passe à l'état $B_1$ ou $B_2$, respectivement
\vfill
Considérons les variables aléatoires $T_A$, \emph{temps passé} dans l'état $A$ (ou temps de séjour dans $A$), $T_{AB_1}$, \emph{temps avant le passage à} $B_1$ et $T_{AB_2}$, \emph{temps avant le passage à} $B_2$
\vfill
Si nous considérons l'état $A$, nous ne pouvons pas observer les variables $T_{AB_1}$ ou $T_{AB_2}$. Ce qui est observable est le temps de séjour dans $A$
\[
T^*_A=\min\left( T_{AB_1},T_{AB_2} \right)
\]
(où $^*$ indique qu'une quantité est observable)
\end{frame}

\begin{frame}{Taux de défaillance par type d'événement}
Nous avons deux (ou plusieurs) types d'événements dont les taux de défaillance individuels doivent être pris en compte
\begin{align*}
h_j(t) &= \lim_{\Delta t\to 0} \frac{\mathbb{P}( T<t+\Delta t, S=S_j | T\geq t)}{\Delta t} 
\end{align*}
où $\mathbb{P}(T<t+\Delta t, S=S_j | T\geq t)$ est la probabilité de défaillance due à la cause $S_j$ ($j=1,2$ ici), c'est-à-dire $S$ est une v.a. discrète représentant l'événement qui se produit
\end{frame}

\begin{frame}
Par la loi de probabilité totale, puisqu'un seul des événements peut avoir lieu, s'il y a $n$ risques, alors
$$
h(t) = \sum_{i=1}^n h_j(t)
$$
ou, de manière identique,
$$
\mathcal{S}(t)
=
\exp\left(
  -\int_0^t \sum\textstyle_{j=1}^n h_j(s)\ ds
\right)
$$
\end{frame}

\begin{frame}
En conséquence, supposons qu'un processus est soumis à deux risques exponentiels concurrents avec des distributions respectives de paramètres $\theta_1$ et $\theta_2$
\vfill
Alors le temps de séjour moyen dans l'état initial avant d'être affecté par l'un des deux risques est
$$
\frac{1}{\theta_1+\theta_2}
$$
\end{frame}

\subsection{Deux distributions « extrêmes » et une plus agréable}
% The section page
\newSubSectionSlide{FIGS-slides-admin/Gemini_Generated_Image_5yvymh5yvymh5yvy.jpeg}

\begin{frame}\frametitle{La distribution exponentielle}
La variable aléatoire $T$ a une distribution \defword{exponentielle} si sa
fonction de densité de probabilité prend la forme
\begin{equation}\label{eq:exp_distrib}
f(t)=\begin{cases}0&\textrm{si }t<0,\\
\theta e^{-\theta t}&\textrm{si }t\geq 0,
\end{cases}
\end{equation}
avec $\theta>0$. Alors la
fonction de survie pour l'état $A$ est de la forme $\S(t)=e^{-\theta
  t}$, pour $t\geq 0$, et le temps de séjour moyen dans l'état $A$ est
\[
\tau=\int_0^\infty e^{-\theta t}dt=\frac 1\theta
\]
\end{frame}

\begin{frame}\frametitle{Particularités de la distribution exponentielle}
L'écart-type d'une distribution exponentielle est également $1/\theta$. Lors de l'estimation de $\theta$, il est impossible de distinguer la moyenne et l'écart-type
\vfill
La distribution exponentielle est \defword{sans mémoire} : sa probabilité conditionnelle obéit à
\[
P(T > s + t\; |\; T > s) = P(T > t),\quad\forall s, t \ge 0
\]

Les distributions exponentielle et géométrique sont les seules distributions de probabilité sans mémoire
\vfill
La distribution exponentielle a une fonction de risque constante $h(t)\equiv\theta$
\end{frame}



\maxFrameImage{FIGS/course-03-pdf-cdf-surv-hazard-expon-1.pdf}

\begin{frame}\frametitle{La distribution delta de Dirac}
Si pour une constante $\omega>0$,
\[
\S(t)=
\left\{
\begin{array}{ll}
1, & 0\leq t\leq\omega \\
0, & \omega<t
\end{array}
\right.
\]
ce qui signifie que $T$ a une distribution delta de Dirac
$\delta_\omega(t)$, alors le temps de séjour moyen est
\[
\tau=\int_0^\omega dt=\omega
\]
avec écart-type $\sigma=0$
\end{frame}

\begin{frame}{La distribution Gamma}
La v.a. $X$ est distribuée selon une loi \defword{Gamma} ($X\,\sim\Gamma(k, \theta)$) avec \defword{paramètre de forme} $k$ et \defword{paramètre d'échelle} $\theta$ (ou \defword{taux} $\beta = 1/\theta$) (tous positifs) si sa fonction de densité de probabilité prend la forme
\begin{equation}\label{eq:gamma_distrib}
f(x;k,\theta) = \frac{x^{k-1} \mathrm{e}^{-\frac{x}{\theta}}}{\Gamma ( k)\theta ^k}
\end{equation}
où $x>0$ et $\Gamma$ est la fonction Gamma d'Euler, définie pour tout $z\in\mathbb{C}$ tel que $\Re(z) > 0$ par
\[
\Gamma : z \mapsto \int_0^{+\infty}  t^{z-1}\,\mathrm{e}^{-t}\,\mathrm{d}t
\]
\end{frame}

\begin{frame}{Propriétés de la distribution Gamma}
Moyenne $k\theta$, variance $k\theta^2$
\vfill
Fonction de survie
\[
\mathcal{S}(t)=
1-\frac{1}{\Gamma(k)}\gamma\left(
k,\frac{t}{\theta} 
\right)
=1-\frac{1}{\Gamma(k)}\gamma\left(
k,\beta t
\right)
\]
où 
\[
\gamma(a,x)=\int_0^x t^{a-1}{\rm e}^{-t}{\rm d}t
\]
est une fonction Gamma incomplète
\end{frame}

\subsection{Un modèle de cohorte simple avec décès} 
% The section page
\newSubSectionSlide{FIGS-slides-admin/Gemini_Generated_Image_5yvymh5yvymh5yvy.jpeg}

\begin{frame}\frametitle{Un modèle pour une cohorte avec une cause de décès}
Considérons une \defword{cohorte} d'individus nés au même moment, par exemple, la même année
\vfill
\begin{itemize}
\item Au temps $t=0$, il y a initialement $N_0>0$ individus
\item Toutes les causes de décès sont combinées ensemble 
\item Le temps jusqu'au décès, pour un individu donné, est une variable aléatoire $T$, avec une distribution de densité de probabilité continue $f(t)$ et une fonction de survie $\S(t)$
\end{itemize}
\vfill
$N(t)$ la population de la cohorte au temps $t\geq 0$
\begin{equation}\label{eq:N_general}
N(t)=N_0\S(t)
\end{equation}
\vfill
$\S(t)$ proportion de la population initiale encore en vie au temps $t$, donc $N_0\S(t)$ nombre dans la cohorte encore en vie au temps $t$
\end{frame}

\begin{frame}\frametitle{Cas où $T$ est distribuée exponentiellement}
Supposons que $T$ ait une distribution exponentielle avec moyenne $1/d$ (ou paramètre $d$), $f(t)=de^{-dt}$. Alors la fonction de survie est $\S(t)=e^{-dt}$, et \eqref{eq:N_general} prend la forme
\begin{equation}\label{eq:N}
N(t)=N_0e^{-dt}
\end{equation}
\vfill
Notez maintenant que
\begin{align*}
\frac{d}{dt} N(t) &= -dN_0e^{-dt} \\
&= -dN(t)
\end{align*}
avec $N(0)=N_0$.
\vfill
{\red $\Rightarrow$} L'EDO $N'=-dN$ fait l'hypothèse que l'espérance de vie à la naissance est distribuée exponentiellement
\end{frame}

\begin{frame}\frametitle{Cas où $T$ a une distribution delta de Dirac}
Supposons que $T$ ait une distribution delta de Dirac en $t=\omega$, donnant la
fonction de survie 
\[
\S(t)=\begin{cases}
1, & 0\leq t\leq\omega \\
0, & t>\omega 
\end{cases}
\]
Alors \eqref{eq:N_general} prend la forme
\begin{equation}\label{eq:N2}
N(t)=\begin{cases}
N_0, & 0\leq t\leq\omega \\
0, & t>\omega
\end{cases}
\end{equation}
Tous les individus survivent jusqu'au temps $\omega$, puis ils meurent tous au temps $\omega$
\vfill
Ici, $N'=0$ partout sauf en $t=\omega$, où elle est indéfinie
\end{frame}

\subsection{Une correction possible au problème de la distribution exponentielle} 
% The section page

\begin{frame}{Survie pour la distribution exponentielle}
\begin{center}

\end{center}

\begin{frame}{Problèmes avec la distribution exponentielle}
\bbullet La survie chute rapidement
\vfill
\bbullet La survie continue bien au-delà de la moyenne
\vfill
Acceptable si ce qui compte est la durée moyenne de séjour dans un compartiment (par ex., dynamique à long terme)
\vfill
Plus délicat si l'on s'intéresse à la dynamique à court terme
\vfill
\bbullet La distribution exponentielle avec paramètre $\theta$ a la même moyenne et écart-type $1/\theta$, c'est-à-dire qu'un seul paramètre contrôle la moyenne et la dispersion autour de la moyenne
\end{frame}

\begin{frame}{Les distributions exponentielles sont « mauvaises » mais aussi cool}
$X_1$ et $X_2$ 2 v.a. i.i.d. (indépendantes et identiquement distribuées) avec paramètres $\theta_1$ et $\theta_2$. Alors la fonction de densité de probabilité de la v.a. $Z=X_1+X_2$ est donnée par la convolution
\begin{align}
 f_Z(z) &= \int_{-\infty}^\infty f_{X_1}(x_1) f_{X_2}(z - x_1)\,dx_1 \nonumber\\
   &= \int_0^z \theta_1 e^{-\theta_1 x_1} \theta_2 e^{-\theta_2(z - x_1)} \, dx_1 \nonumber\\
   &= \theta_1 \theta_2 e^{-\theta_2 z} \int_0^z e^{(\theta_2 - \theta_1)x_1}\,dx_1 \nonumber\\
   &= \begin{cases}
        \dfrac{\theta_1 \theta_2}{\theta_2-\theta_1} \left(e^{-\theta_1 z} - e^{-\theta_2 z}\right) & \text{ si } \theta_1 \neq \theta_2 \\
        \theta^2 z e^{-\theta z} & \text{ si } \theta_1 = \theta_2 =: \theta
      \end{cases}
\end{align}
\end{frame}

\begin{frame}{L'outil que nous utilisons}
\begin{theorem}
Soit $X_i$ des variables aléatoires indépendantes distribuées exponentiellement avec paramètre $\xi$ et $Y = \sum_{i=1}^n X_i$
\vskip1cm
Alors la variable aléatoire $Y\rightsquigarrow E(n,\xi)$, une distribution d'Erlang avec paramètre de \emph{forme} $n$ et paramètre d'\emph{échelle} $\xi$
\end{theorem}
\vfill
(Distribution d'Erlang : distribution Gamma avec paramètre de forme entier)
\end{frame}

\subsection{Temps de séjour dans un modèle de transmission de maladie SIS} 
% The section page
\newSubSectionSlide{FIGS-slides-admin/Gemini_Generated_Image_5yvymh5yvymh5yvy.jpeg}

\begin{frame}\frametitle{Un modèle SIS}
\framesubtitle{Hypothèses}
\begin{itemize}
\item Les individus se rétablissent typiquement de la maladie
\vfill
\item La maladie ne confère pas d'immunité
\vfill
\item Il n'y a pas de naissance ou de décès (de la maladie ou naturel) \newline\imply\;
Population totale constante $N\equiv N(t)=S(t)+I(t)$
\vfill
\item L'infection est de type \defword{incidence standard}
\end{itemize}
\end{frame}


\begin{frame}\frametitle{Rétablissement}
\begin{itemize}
\item Les modèles traditionnels supposent que le rétablissement se produit avec un taux constant
$\gamma$
\vfill
\item Ici, parmi les individus qui deviennent
infectieux au temps $t_0$, une fraction $\S(t-t_0)$ reste infectieuse au
temps $t\geq t_0$
\vfill
\item \imply\;
Pour $t\geq 0$, $\S(t)$ est une fonction de survie. En tant que telle, elle vérifie
$\S(0)=1$ et $\S$ est non négative et non croissante
\end{itemize}
\end{frame}


\begin{frame}\frametitle{Modèle pour les individus infectieux}
Puisque $N$ est constant, $S(t)=N-I(t)$ et nous n'avons besoin de
considérer que l'équation suivante (où $S$ est utilisé pour plus de clarté)
\begin{equation}
I(t) = I_0(t)+ \int_0^t\beta\frac{S(u)I(u)}{N} \S(t-u) du
\label{eq:SIS_I} 
\end{equation}
\vfill
\begin{itemize}
\item $I_0(t)$ nombre d'individus qui étaient infectieux au temps
$t=0$ et le sont toujours au temps $t$
\begin{itemize}
\item $I_0(t)$ est non négatif, non croissant, et
tel que $\lim_{t\to\infty}I_0(t)=0$
\end{itemize}
\item $\S(t-u)$ proportion d'individus qui sont devenus infectieux au temps $u$ et qui le sont toujours au temps $t$
\end{itemize}
\end{frame}


\begin{frame}\frametitle{Expression sous l'intégrale}
Équation intégrale pour le nombre d'individus infectieux : 
\begin{equation}
I(t) = I_0(t)+ \int_0^t\beta\frac{(N-I(u))I(u)}{N} \S(t-u) du
\tag{\ref{eq:SIS_I}} 
\end{equation}
Le terme
\[
\beta\frac{(N-I(u))I(u)}{N} \S(t-u)
\]
\begin{itemize}
\item $\beta (N-I(u))I(u)/N$ est le taux auquel de nouveaux infectieux sont créés, au temps $u$
\item multiplier par $\S(t-u)$ donne la proportion de ceux qui sont devenus
infectieux au temps $u$ et qui le sont toujours au temps $t$
\end{itemize}
Sommer sur $[0,t]$ donne le nombre d'individus infectieux au temps $t$
\end{frame}


\begin{frame}\frametitle{Cas d'un temps de rétablissement distribué exponentiellement}
Supposons $\S(t)$ tel que le temps de séjour dans l'état infectieux
ait une distribution exponentielle avec moyenne $1/\gamma$,
\textit{c'est-à-dire}, $\S(t)=e^{-\gamma t}$
\vfill
La fonction de condition initiale $I_0(t)$ prend la forme
\[
I_0(t)=I_0(0)e^{-\gamma t}
\]
avec $I_0(0)$ le nombre d'individus infectieux au temps $t=0$. Obtenu en considérant la cohorte d'individus initialement infectieux, donnant un
modèle tel que \eqref{eq:N_general}
\vfill
L'équation (\ref{eq:SIS_I}) devient
\begin{equation}\label{eq:I_ODE}
I(t)=I_0(0)e^{-\gamma t}+\int_0^t \beta\frac{(N-I(u))I(u)}{N} e^{-\gamma
(t-u)}du
\end{equation}
\end{frame}

\begin{frame}
Prendre la dérivée temporelle de \eqref{eq:I_ODE} donne
\begin{align*}
I'(t) &= -\gamma I_0(0)e^{-\gamma t}-\gamma\int_0^t
\beta\frac{(N-I(u))I(u)}{N}e^{-\gamma(t-u)}du \\
&\quad +\beta \frac{(N-I(t))I(t)}{N} \\
&= -\gamma\left(I_0(0)e^{-\gamma t}+
\int_0^t \beta\frac{(N-I(u))I(u)}{N}e^{-\gamma(t-u)}du\right) \\
&\quad +\beta \frac{(N-I(t))I(t)}{N} \\
&= \beta \frac{(N-I(t))I(t)}{N}-\gamma I(t)
\end{align*}
\vfill
Ceci est l'équation différentielle ordinaire (EDO) classique de type logistique pour $I$ dans un modèle SIS sans dynamique vitale (pas de naissance ou de décès)
\end{frame}



\begin{frame}\frametitle{Cas d'une fonction de survie en escalier}
Considérons le cas où le temps passé infecté a une fonction de survie
\[
\S(t)=\begin{cases}
1, & 0\leq t\leq\omega,\\
0, & t>\omega.
\end{cases}
\]
c'est-à-dire, le temps de séjour dans l'état infectieux est une constante
$\omega>0$
\vfill
Dans ce cas (\ref{eq:SIS_I}) devient
\begin{equation}\label{eq:I_DDE}
I(t)=I_0(t)+\int_{t-\omega}^t \beta\frac{(N-I(u))I(u)}{N} du.
\end{equation}
Ici, il est plus difficile d'obtenir une expression pour $I_0(t)$. Il est cependant supposé que $I_0(t)$ s'annule pour $t>\omega$
\end{frame}

\begin{frame}
Lorsque différentiée, \eqref{eq:I_DDE} donne, pour $t\geq\omega$,
\[
I'(t)=I_0'(t)+\beta\frac{(N-I(t))I(t)}{N}
-\beta\frac{\left(N-I(t-\omega)\right)I(t-\omega)}{N}.
\]
Puisque $I_0(t)$ s'annule pour $t>\omega$, ceci donne l'équation différentielle à retard (EDR)
\[
I'(t)=\beta\frac{(N-I(t))I(t)}{N}
-\beta\frac{(N-I(t-\omega))I(t-\omega)}{N}.
\]
\end{frame}

\begin{frame}\frametitle{Hypothèses sur $\S$}
$\S(t)$ est une fonction non négative et non croissante avec $\S(0^+)=1$, et telle que $\int_0^\infty \S(u)du$ est positive et finie
\vfill
Donc $\S(t)$ est une fonction de survie
\end{frame}

\begin{frame}\frametitle{Le modèle SIS avec vaccination}

\begin{align}
  \frac{dI(t)}{dt}&= \beta(S(t)+\sigma V(t))I(t)-(d+\gamma)I(t)
  \label{sys:SIVS_general_I}\\
  V(t)&= V_0(t)
  +\int_0^t(\phi S(u)+\alpha d)\S(t-u)e^{-d(t-u)}
  e^{-\sigma\beta\int_u^tI(x)dx}du \label{sys:SIVS_general_V}
\end{align}
\end{subequations}
\vfill
\begin{itemize}
\item $\alpha d$ proportion de nouveau-nés vaccinés
\item $\phi S(u)$ proportion de susceptibles vaccinés
\item $\S(t-u)$ fraction de la proportion vaccinée encore dans la classe $V$
  $t-u$ unités de temps après y être entrée
\item $e^{-d(t-u)}$ fraction de la proportion vaccinée
pas décédée de causes naturelles
\item $e^{-\sigma\beta\int_u^t I(x)dx}$ fraction de la proportion
  vaccinée pas passée dans la classe infectieuse
\end{itemize}
\end{frame}

\begin{frame}{Soyez conscient (méfiez-vous ?) de ce qui se cache sous le capot}
Les EDO et les EDR à retard discret vont bien, mais elles cachent des hypothèses assez fortes
\vfill
Si vous utilisez une EDO pour un modèle \emph{endémique}, bien : vous travaillez typiquement avec $t\to\infty$ et l'EDO « utilise » la moyenne de votre paramètre
\vfill
Si, d'autre part, vous utilisez une EDO pour évaluer le comportement pour des temps qui ne sont pas trop différents de la moyenne de vos paramètres, alors méfiez-vous
\vfill
Les retards discrets font l'hypothèse que chaque individu dans votre population a le même temps de séjour si un compartiment quelconque se voit appliquer un retard... Vous pourriez vouloir penser aux retards distribués ou à d'autres mécanismes pour compenser ceci
\vfill
Vous pourriez vouloir utiliser un marteau différent
\end{frame}

\subsection{Chaînes de Markov en temps continu}
\newSubSectionSlide{FIGS-slides-admin/Gemini_Generated_Image_9agynl9agynl9agy.jpeg}
\begin{frame}
    \frametitle{Du temps discret au temps continu}
    \begin{block}{Chaînes de Markov en temps discret (CMTD)}
        Un système transite entre états à des pas de temps fixes et discrets ($n=0, 1, 2, \dots$)
        \begin{itemize}
            \item Le futur dépend uniquement de l'état présent (Propriété de Markov)
            \item Régi par une \textbf{matrice de probabilité de transition} $P$, où $P_{ij}$ est la probabilité de passer de l'état $i$ à $j$ en une étape
        \end{itemize}
    \end{block}

    \begin{block}{Chaînes de Markov en temps continu (CMTC)}
        Un système peut transiter entre états à \textbf{n'importe quel moment}
        \begin{itemize}
            \item Le temps passé dans un état est une \textbf{variable aléatoire continue}
            \item Le « temps de séjour » dans tout état $i$ suit une \textbf{distribution exponentielle} paramétrée par un \textit{taux de sortie} $q_i$
            \item Ceci est une conséquence directe de la Propriété de Markov appliquée au temps continu (l'exponentielle est la seule distribution continue qui est « sans mémoire »)
        \end{itemize}
    \end{block}
\end{frame}

\begin{frame}
    \frametitle{Taux de transition}
    Dynamique d'une CMTC définie par des \textbf{taux de transition}, pas des probabilités

    \begin{definition}[Taux de transition]
        Pour deux états $i \neq j$, le taux $q_{ij} \ge 0$ est le taux instantané de transition de l'état $i$ à l'état $j$
        \begin{itemize}
            \item Pour un petit intervalle de temps $\Delta t$, la probabilité de transition de $i$ à $j$ est approximativement $q_{ij} \Delta t$
            \item Le \defword{taux de sortie} total de l'état $i$ est $q_i = \sum_{j \neq i} q_{ij}$
            \item Le temps passé dans l'état $i$ est une variable aléatoire exponentielle $T_i \sim \text{E}(q_i)$
        \end{itemize}
    \end{definition}
\end{frame}

\begin{frame}{La matrice génératrice}
    \begin{block}{Matrice génératrice (Matrice Q)}
        Rassembler tous les taux de transition dans une seule matrice $Q$
        \begin{itemize}
            \item Hors diagonale : $Q_{ij} = q_{ij}$ pour $i \neq j$ (Le taux de passage de $i$ à $j$)
            \item Diagonale : $Q_{ii} = -q_i = -\sum_{j \neq i} q_{ij}$. (Le négatif du taux de sortie total de $i$)
        \end{itemize}
        Une propriété clé est que toutes les lignes de $Q$ somment à zéro : $\sum_{j} Q_{ij} = 0$.
    \end{block}

\end{frame}


\begin{frame}
    \frametitle{Équations de Kolmogorov}
    Soit $P(t)$ la matrice où $P_{ij}(t) = \mathbb{P}(X(t)=j | X(0)=i)$. Comment $P(t)$ évolue-t-elle dans le temps ?
    \vspace{1em}

    \begin{alertblock}{Équations de Kolmogorov progressives}
        Décrit le taux de changement de probabilité de finir dans un état cible $j$
        $$ \frac{d}{dt}P(t) = P(t)Q $$
        Sous forme élémentaire :
        $$ P'_{ij}(t) = \sum_{k} P_{ik}(t) Q_{kj} $$
    \end{alertblock}
\end{frame}

\begin{frame}{Solution de l'ÉKP}
        La solution est l'\textbf{exponentielle matricielle}
        $$ P(t) = e^{tQ} = \sum_{k=0}^{\infty} \frac{(tQ)^k}{k!} $$
        \vfill
        La matrice génératrice $Q$ « génère » l'évolution du processus
\end{frame}

\begin{frame}{L'algorithme de Gillespie}
    \begin{itemize}
        \item Aussi appelé l'algorithme de simulation stochastique (SSA)
        \vfill
        \item Dérivé en 1976 par Daniel Gillespie
        \vfill
        \item Génère des solutions possibles pour les CMTC
        \vfill
        \item Extrêmement simple, donc vaut la peine d'apprendre à implémenter ; il existe cependant des paquets que vous pouvez utiliser (voir plus tard)
    \end{itemize}
\end{frame}


\begin{frame}{L'algorithme de Gillespie}
Supposons que le système a un état $\mathbf{x}(t)$ avec condition initiale $\mathbf{x}(t_0)=\mathbf{x}_0$ et \emph{fonctions de propension} $a_i$ des réactions élémentaires
\vfill
définir $t\leftarrow t_0$ et $\mathbf{x}(t)\leftarrow \mathbf{x}_0$\\
tant que {$t\leq t_f$}\\
-\qquad $\xi_t\leftarrow \sum_j a_j(\mathbf{x}(t))$\\
-\qquad Tirer $\tau_t$ de $T\thicksim \mathcal{E}(\xi_t)$\\
-\qquad Tirer $\zeta_t$ de $\mathcal{U}([0,1])$\\
-\qquad Trouver $r$, le plus petit entier tel que $\sum_{k=1}^j a_k(\mathbf{x}(t))> \zeta_t\sum_j a_j(\mathbf{x}(t))=\zeta_t\xi_t$\\
-\qquad Effectuer la prochaine réaction (celle indexée $r$)\\
-\qquad $t\leftarrow t+\tau_t$\\
\end{frame}


\begin{frame}{Tirer aléatoirement d'une distribution exponentielle}
    Si vous n'avez pas de générateur de nombres aléatoires de distribution exponentielle.. Nous voulons $\tau_t$ de $T\thicksim\mathcal{E}(\xi_t)$, c'est-à-dire, $T$ a une fonction de densité de probabilité
    $$
    f(x,\xi_t)=
    \xi_te^{-\xi_t x}\mathbf{1}_{x\geq 0}
    $$
    Utiliser la fonction de distribution cumulative $F(x,\xi_t)=\int_{-\infty}^x f(s,\xi_t)\,ds$
    $$
    F(x,\xi_t)=
    (1-e^{-\xi_t x})\mathbf{1}_{x\geq 0}
    $$
    qui a des valeurs dans $[0,1]$. Donc tirer $\zeta$ de $\mathcal{U}([0,1])$ et résoudre $F(x,\xi_t)=\zeta$ pour $x$
    \begin{align*}
    F(x,\xi_t)=\zeta & \Leftrightarrow 1-e^{-\xi_tx}=\zeta \\
    &\Leftrightarrow e^{-\xi_tx} = 1-\zeta \\
    &\Leftrightarrow \xi_tx = -\ln(1-\zeta) \\
    &\Leftrightarrow \boxed{x = \frac{-\ln(1-\zeta)}{\xi_t}}
    \end{align*}
\end{frame}


\begin{frame}{L'algorithme de Gillespie (modèle SIS avec seulement l'éq. I)}
définir $t\leftarrow t_0$ et $I(t)\leftarrow I(t_0)$\\
tant que {$t\leq t_f$}\\
-\qquad $\xi_t\leftarrow \beta (P^\star-i)i+\gamma i$\\
-\qquad Tirer $\tau_t$ de $T\thicksim \mathcal{E}(\xi_t)$\\
-\qquad $v\leftarrow\left[\beta (P^\star-i)i,\xi_t\right]/\xi_t$\\
-\qquad Tirer $\zeta_t$ de $\mathcal{U}([0,1])$\\
-\qquad Trouver $pos$ tel que $v_{pos-1}\leq\zeta_t\leq v_{pos}$\\
-\qquad selon {$pos$}\\
\qquad\qquad - 1: Nouvelle infection, $I(t+\tau_t)=I(t)+1$ \\
\qquad\qquad - 2: Fin de la période infectieuse, $I(t+\tau_t)=I(t)-1$ \\
- $t\leftarrow t+\tau_t$
\end{frame}


\begin{frame}{Parfois Gillespie tourne mal}
    \begin{itemize}
        \item Rappelons que le temps inter-événements est distribué exponentiellement
        \item Étape critique de l'algorithme de Gillespie :
        \begin{itemize}
            \item $\xi_t\leftarrow$ poids de tous les événements possibles (\emph{propension})
            \item Tirer $\tau_t$ de $T\thicksim \mathcal{E}(\xi_t)$
        \end{itemize}
        \item Donc le temps inter-événements $\tau_t\to 0$ si $\xi_t$ devient très grand pour un certain $t$
        \item Ceci peut causer le ralentissement de la simulation jusqu'à l'arrêt
    \end{itemize}
\end{frame}

        \item Individus meurent à un taux \emph{per capita} $d$
        \item Implémentons cela en utilisant le Gillespie classique
    \end{itemize}

\section{Approximations par processus de branchement des CMTC}
% The section page
\newSectionSlide{FIGS-slides-admin/Gemini_Generated_Image_5yvymh5yvymh5yvy.jpeg}

\begin{frame}
    \frametitle{Qu'est-ce qu'un processus de branchement ?}
    \begin{block}{L'idée centrale}
        Un processus de branchement est un modèle mathématique pour une population où les individus produisent un nombre aléatoire de descendants puis meurent.
    \end{block}

    \begin{itemize}
        \item Pensez aux bactéries qui se divisent, à un virus qui se propage, ou même à la survie des noms de famille.
        \item Nous commençons avec une population initiale, $Z_0$.
        \item Chaque individu de la génération $n$ produit un nombre de descendants pour la génération $n+1$.
        \item Ce « nombre de descendants » est une variable aléatoire. Tous les individus produisent des descendants selon la même distribution de probabilité, indépendamment les uns des autres.
    \end{itemize}
\end{frame}

\begin{frame}
    \frametitle{Le processus de Galton-Watson}

    Soit $Z_n$ la taille de la population à la génération $n$. Nous commençons typiquement avec $Z_0 = 1$.

    La population évolue selon la règle :
    $$ Z_{n+1} = \sum_{i=1}^{Z_n} X_{n,i} $$

    \begin{itemize}
        \item Le terme $X_{n,i}$ représente le nombre de descendants produits par le $i$-ème individu de la génération $n$.
        \item Les variables $\{X_{n,i}\}$ sont supposées être des variables aléatoires \textbf{indépendantes et identiquement distribuées (i.i.d.)} à valeurs entières.
        \item Nous appelons leur distribution commune $\{p_k\}_{k=0}^\infty$ la \textbf{distribution des descendants}, où $p_k = P(X = k)$.
    \end{itemize}

    \begin{alertblock}{Les questions fondamentales}
    \begin{enumerate}
        \item Quelle est la taille espérée à long terme de la population ?
        \item Quelle est la probabilité que la population finisse par s'éteindre ?
    \end{enumerate}
    \end{alertblock}
\end{frame}

\begin{frame}
    \frametitle{Moyenne des descendants}

    Le destin de la population dépend d'un seul paramètre : la moyenne de la distribution des descendants

    $$ \mu = E[X] = \sum_{k=0}^{\infty} k \cdot p_k $$

    \begin{block}{Taille de population espérée}
        En utilisant la loi de l'espérance totale, nous trouvons la taille espérée de la prochaine génération :
        $$ E[Z_{n+1} | Z_n] = E\left[\sum_{i=1}^{Z_n} X_{n,i} \bigg| Z_n\right] = Z_n E[X] = Z_n \mu $$

        En prenant l'espérance à nouveau, nous obtenons une récurrence simple :
        $$ E[Z_{n+1}] = \mu E[Z_n] $$

        Ceci implique :
        $$ E[Z_n] = Z_0 \mu^n $$
    \end{block}
\end{frame}

% --- SLIDE 5: The Three Regimes of Population Growth ---
\begin{frame}
    \frametitle{Les trois régimes de croissance de population}

    Le comportement de $E[Z_n] = Z_0 \mu^n$ suggère trois cas distincts :

    \begin{columns}[T]
        \begin{column}{0.33\textwidth}
            \begin{block}{Sous-critique ($\mu < 1$)}
                $E[Z_n] \to 0$. La population devrait rétrécir. Elle s'éteint avec probabilité 1.
            \end{block}
        \end{column}
        \begin{column}{0.33\textwidth}
            \begin{block}{Critique ($\mu = 1$)}
                $E[Z_n] = Z_0$. La population devrait rester stable. Curieusement, elle s'éteint quand même avec probabilité 1.
            \end{block}
        \end{column}
        \begin{column}{0.33\textwidth}
            \begin{block}{Sur-critique ($\mu > 1$)}
                $E[Z_n] \to \infty$. La population devrait croître exponentiellement. Elle a une probabilité non nulle de survivre à jamais.
            \end{block}
        \end{column}
    \end{columns}
\end{frame}

% --- SLIDE 6: Calculating the Extinction Probability ---
\begin{frame}
    \frametitle{Outil : La fonction génératrice de probabilité}

    Pour trouver la probabilité d'extinction, nous avons besoin d'un outil puissant : la \textbf{fonction génératrice de probabilité (FGP)} de la distribution des descendants $X$.

    $$ G(s) = E[s^X] = \sum_{k=0}^{\infty} p_k s^k \quad \text{pour } |s| \le 1 $$

    \begin{block}{Propriétés clés}
    \begin{itemize}
        \item $G(1) = \sum p_k = 1$
        \item La moyenne peut être trouvée à partir de la dérivée : $G'(1) = \sum k p_k = \mu$.
        \item La FGP de $Z_n$ est le $n$-ième itéré de $G(s)$ avec elle-même. Si $G_n(s)$ est la FGP de $Z_n$, alors $G_{n+1}(s) = G(G_n(s))$.
    \end{itemize}
    \end{block}

\end{frame}

% --- SLIDE 7: The Extinction Probability Equation ---
\begin{frame}
    \frametitle{Équation de la probabilité d'extinction}
    Soit $\pi_0$ la probabilité d'extinction éventuelle, en commençant avec $Z_0=1$.
    $$ \pi_0 = P(\text{la population s'éteint}) = \lim_{n \to \infty} P(Z_n = 0) $$

    Puisque $P(Z_n=0) = G_n(0)$, et $G_{n+1}(0) = G(G_n(0))$, à la limite la probabilité d'extinction $\pi_0$ doit satisfaire l'équation :

    \begin{center}
        \Huge
        $ \pi_0 = G(\pi_0) $
    \end{center}



    \begin{theorem}
    La probabilité d'extinction $\pi_0$ est la \textbf{plus petite solution non négative} de l'équation $s = G(s)$.
    \end{theorem}

    \begin{itemize}
        \item Si $\mu \le 1$, la seule solution dans $[0,1]$ est $s=1$. Donc $\pi_0=1$.
        \item Si $\mu > 1$, il existe une solution unique dans $[0,1)$, qui est la probabilité d'extinction $\pi_0 < 1$.
    \end{itemize}
\end{frame}

% --- SLIDE 8: From Discrete to Continuous Time ---
\begin{frame}
    \frametitle{Du temps discret au temps continu}

    \begin{block}{Limitation de Galton-Watson}
        Les générations n'arrivent pas en étapes synchronisées dans le monde réel. Les individus donnent naissance et meurent à des moments aléatoires.
    \end{block}

    Cela nous amène aux \textbf{Chaînes de Markov en temps continu (CMTC)}.

    \begin{itemize}
        \item L'état du système est la taille de la population, $k \in \{0, 1, 2, \dots\}$.
        \item Au lieu de générations, nous avons des taux de transition :
            \begin{itemize}
                \item $\lambda_k$ : taux de naissance lorsque la population est de taille $k$ (passe à $k+1$).
                \item $\delta_k$ : taux de décès lorsque la population est de taille $k$ (passe à $k-1$).
            \end{itemize}
        \item Souvent, nous supposons que ces taux sont linéaires : $\lambda_k = k\lambda$ et $\delta_k = k\delta$. Cela signifie que les individus agissent indépendamment.
    \end{itemize}
\end{frame}

% --- SLIDE 9: Branching Process Approximation of a CTMC ---
\begin{frame}
    \frametitle{Approximation par processus de branchement d'une CMTC}

    \begin{alertblock}{L'idée clé}
    Au début d'une épidémie (ou pour une très grande population), la dynamique causée par un seul individu est largement indépendante des autres.
    \end{alertblock}

    Cela nous permet d'approximer le début d'un processus de population CMTC avec un processus de branchement.

    \textbf{Exemple : Une épidémie simple (modèle SIR)}
    \begin{itemize}
        \item $S$ : Susceptible, $I$ : Infecté, $R$ : Rétabli.
        \item Une personne infectée rencontre d'autres personnes à un certain taux. Si elle rencontre un susceptible, une nouvelle infection peut se produire (un « descendant »).
        \item La personne infectée se rétablit (ou meurt) à un autre taux, terminant sa période infectieuse.
        \item \textbf{Question :} Combien de nouvelles infections une seule personne infectée cause-t-elle en moyenne ?
    \end{itemize}
\end{frame}

% --- SLIDE 10: Case Study: The Basic Reproduction Number $\mathcal{R}_0$ ---
\begin{frame}
    \frametitle{Étude de cas : Le nombre de reproduction de base $\mathcal{R}_0$}

    Considérons un seul individu infecté dans une grande population de susceptibles.
    \begin{itemize}
        \item Soit $\beta$ le taux d'infection (taux de production de « descendants »).
        \item Soit $\gamma$ le taux de rétablissement (taux de « mort »).
    \end{itemize}

    La durée de vie infectieuse de l'individu est une variable aléatoire exponentielle avec moyenne $1/\gamma$.



    Le nombre moyen d'infections secondaires qu'il cause est :
    $$ \mathcal{R}_0 = (\text{taux d'infection}) \times (\text{période infectieuse moyenne}) = \beta \times \frac{1}{\gamma} = \frac{\beta}{\gamma} $$

    \begin{block}{La connexion}
    $\mathcal{R}_0$ est précisément le \textbf{nombre moyen de descendants} $\mu$ pour le processus de branchement intégré qui approxime le début de l'épidémie.
    \end{block}
\end{frame}

% --- SLIDE 11: Applying Branching Theory to Epidemics ---
\begin{frame}
    \frametitle{Appliquer la théorie de branchement aux épidémies}

    Le destin de la phase initiale de l'épidémie est déterminé par $\mathcal{R}_0$ :

    \begin{itemize}
        \item Si $\mathcal{R}_0 \le 1$ ($\mu \le 1$) : Le nombre d'individus infectés est un processus sous-critique ou critique. L'épidémie s'éteindra avec probabilité 1.

        \item Si $\mathcal{R}_0 > 1$ ($\mu > 1$) : Le processus est sur-critique. Il y a une probabilité positive que l'épidémie décolle et cause une épidémie majeure.
    \end{itemize}



    Nous pouvons même calculer la probabilité d'une épidémie majeure ! C'est $1 - \pi_0$, où $\pi_0$ est la probabilité d'extinction.

    Pour ce processus d'infection naissance-mort simple, la FGP est $G(s) = \frac{\gamma}{\beta+\gamma} + \frac{\beta}{\beta+\gamma}s$. Résoudre $s=G(s)$ donne la probabilité d'extinction :

    $$ \pi_0 = \frac{\gamma}{\beta} = \frac{1}{\mathcal{R}_0} $$

    La probabilité d'une épidémie majeure est $1 - 1/\mathcal{R}_0$.
\end{frame}

% --- SLIDE 12: Summary ---
\begin{frame}
    \frametitle{Résumé}

    \begin{itemize}
        \item Les \textbf{processus de branchement} modélisent les populations avec une génération de descendants i.i.d.

        \item Le destin de la population est déterminé par le \textbf{nombre moyen de descendants} $\mu$. L'extinction est certaine si $\mu \le 1$.

        \item La \textbf{probabilité d'extinction} $\pi_0$ peut être calculée comme le plus petit point fixe non négatif de la fonction génératrice de probabilité $G(s)$.

        \item Les stades initiaux de nombreuses \textbf{chaînes de Markov en temps continu} à grande échelle peuvent être approximés par un processus de branchement.

        \item Cela nous permet d'appliquer la théorie à des problèmes du monde réel, comme le calcul du \textbf{nombre de reproduction de base $\mathcal{R}_0$} d'une épidémie et sa probabilité de causer une épidémie majeure.
    \end{itemize}
\end{frame}
